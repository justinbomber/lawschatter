-- ========================================
-- Schema & Extensions (Supabase friendly)
-- ========================================
create schema if not exists lawschatter;

create extension if not exists pgcrypto; -- gen_random_uuid()
create extension if not exists pg_trgm;  -- trigram模糊搜尋（GIN + gin_trgm_ops）

-- ========================================
-- Helper: touch updated_at
-- ========================================
create or replace function lawschatter.touch_updated_at()
returns trigger
language plpgsql
as $fn$
begin
  new.updated_at := now();
  return new;
end
$fn$;

-- ========================================
-- Enum types (放在 lawschatter schema 下)
-- ========================================
do $$
begin
  if not exists (
    select 1 from pg_type t
    join pg_namespace n on n.oid = t.typnamespace
    where t.typname = 'sender_role' and n.nspname = 'lawschatter'
  ) then
    execute $$create type lawschatter.sender_role as enum ('user','assistant')$$;
  end if;
end
$$;

-- ========================================
-- USERS (Supabase Auth 模式)
-- 以 auth.users.id 為唯一使用者ID；profiles 為自家擴充
-- ========================================
create table if not exists lawschatter.profiles (
  id            uuid primary key,        
  username      citext unique,     		 
  display_name  varchar(255),            
  avatar_url    text,
  is_active     boolean not null default true,
  created_at    timestamptz not null default now(),
  updated_at    timestamptz not null default now(),
  last_login    timestamptz,

  constraint fk_profiles_auth
    foreign key (id) references auth.users(id) on delete cascade
);

-- ====== Table 說明 ======
comment on table lawschatter.profiles is '使用者擴充資料表（綁定 Supabase auth.users），用來存放應用層所需的 Profile 資訊';

-- ====== 欄位說明 ======
comment on column lawschatter.profiles.id is '與 auth.users.id 綁定的唯一識別碼（UUID）';
comment on column lawschatter.profiles.username is '自訂登入代稱（非必填），大小寫不敏感且唯一';
comment on column lawschatter.profiles.display_name is '使用者顯示名稱，可用於前端顯示';
comment on column lawschatter.profiles.avatar_url is '使用者頭像 URL';
comment on column lawschatter.profiles.is_active is '帳號是否啟用（預設 true）';
comment on column lawschatter.profiles.created_at is '建立時間（預設為 now()）';
comment on column lawschatter.profiles.updated_at is '最後更新時間（更新時由 Trigger 自動維護）';
comment on column lawschatter.profiles.last_login is '最後登入時間，由應用程式在登入時更新';

-- 大小寫不敏感唯一（避免 John / john 重覆)

drop trigger if exists trg_profiles_updated_at on lawschatter.profiles;
create trigger trg_profiles_updated_at

-- ========================================
-- Conversations
-- ========================================
CREATE TABLE IF NOT EXISTS lawschatter.conversations (
  id               uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id          uuid NOT NULL REFERENCES lawschatter.profiles(id) ON DELETE CASCADE,
  title            varchar(500),
  message_count    integer NOT NULL DEFAULT 0,   -- 由 messages 觸發器維護（或應用程式手動維護）
  last_message_at  timestamptz,                  -- 由 messages 觸發器維護（或應用程式手動維護）
  created_at       timestamptz NOT NULL DEFAULT now(),
  updated_at       timestamptz NOT NULL DEFAULT now(),
  CONSTRAINT chk_title_len CHECK (title IS NULL OR length(title) <= 500)
);

COMMENT ON TABLE lawschatter.conversations
  IS '對話主表：儲存每位使用者的對話概要（列表/排序/快速顯示）。';

COMMENT ON COLUMN lawschatter.conversations.id
  IS '對話唯一識別碼（UUID）。';

COMMENT ON COLUMN lawschatter.conversations.user_id
  IS '對話擁有者（對應 profiles.id）。RLS 與多租戶過濾依此欄位。';

COMMENT ON COLUMN lawschatter.conversations.title
  IS '對話標題（可為 NULL）。上限 500 字元以保護前端 UI 版面。';

COMMENT ON COLUMN lawschatter.conversations.message_count
  IS '對話內訊息數（預設 0）。建議由 messages 的 AFTER INSERT/DELETE 觸發器自動維護，避免昂貴 COUNT(*)。';

COMMENT ON COLUMN lawschatter.conversations.last_message_at
  IS '最後一則訊息時間。建議由 messages 的 AFTER INSERT/DELETE 觸發器更新/重算，用於列表排序。';

COMMENT ON COLUMN lawschatter.conversations.created_at
  IS '建立時間（預設 now()）。';

COMMENT ON COLUMN lawschatter.conversations.updated_at
  IS '最後更新時間。由 BEFORE UPDATE 觸發器 touch_updated_at() 自動維護。';

COMMENT ON CONSTRAINT chk_title_len ON lawschatter.conversations
  IS '限制標題長度 <= 500，避免過長字串造成 UI 破版。';

-- 索引：我的對話清單 → 依最後訊息時間排序（NULL 放最後）
CREATE INDEX IF NOT EXISTS idx_conversations_user_lastmsg
  ON lawschatter.conversations (user_id, last_message_at DESC NULLS LAST);
COMMENT ON INDEX lawschatter.idx_conversations_user_lastmsg
  IS '清單常用索引：依 user_id 過濾並以 last_message_at DESC 排序。';

-- 後備排序（必要時用 updated_at）
CREATE INDEX IF NOT EXISTS idx_conversations_user_updated
  ON lawschatter.conversations (user_id, updated_at DESC);
COMMENT ON INDEX lawschatter.idx_conversations_user_updated
  IS '後備索引：依 user_id 過濾並以 updated_at DESC 排序。';

-- updated_at 觸發器
DROP TRIGGER IF EXISTS trg_conversations_updated_at ON lawschatter.conversations;
CREATE TRIGGER trg_conversations_updated_at
BEFORE UPDATE ON lawschatter.conversations
FOR EACH ROW EXECUTE FUNCTION lawschatter.touch_updated_at();

*************************
-- 後端在「寫入新訊息成功」時記得同步
update lawschatter.conversations
set last_message_at = now()
where id = :cid;

************************

-- ========================================
-- roles：用 role_code 代表身分
-- ========================================
CREATE TABLE IF NOT EXISTS lawschatter.roles (
  id           uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  role_code    text NOT NULL UNIQUE,                      -- 例如 'user','assistant','system','tool'
  role_name    text,                                      -- 顯示名稱
  description  text,
  created_at   timestamptz NOT NULL DEFAULT now(),
  updated_at   timestamptz NOT NULL DEFAULT now(),
  CONSTRAINT ck_roles_code_lower CHECK (role_code = lower(role_code))
);

COMMENT ON TABLE  lawschatter.roles IS '系統角色表（取代 enum，支援彈性擴充；供 messages.sender_role_code 參照）';
COMMENT ON COLUMN lawschatter.roles.id          IS '主鍵 UUID。';
COMMENT ON COLUMN lawschatter.roles.role_code   IS '角色代碼（唯一、邏輯使用；一律小寫，例如 user/assistant/system/tool）。';
COMMENT ON COLUMN lawschatter.roles.role_name   IS '角色顯示名稱。';
COMMENT ON COLUMN lawschatter.roles.description IS '角色說明。';
COMMENT ON COLUMN lawschatter.roles.created_at  IS '建立時間。';
COMMENT ON COLUMN lawschatter.roles.updated_at  IS '最後更新時間。';
COMMENT ON CONSTRAINT ck_roles_code_lower ON lawschatter.roles
  IS '強制 role_code 為小寫，避免大小寫混用造成 FK/查詢混亂。';
  
-- updated_at 觸發器
DROP TRIGGER IF EXISTS trg_roles_updated_at ON lawschatter.roles;
CREATE TRIGGER trg_roles_updated_at
BEFORE UPDATE ON lawschatter.roles
FOR EACH ROW EXECUTE FUNCTION lawschatter.touch_updated_at();

-- ========================================
-- messages（改用 role_code 做外鍵）
-- ========================================
create table if not exists lawschatter.messages (
  id               uuid primary key default gen_random_uuid(),
  conversation_id  uuid not null
                   references lawschatter.conversations(id) on delete cascade,
  sender_role_code text not null
                   references lawschatter.roles(role_code) on delete restrict,
  content          text not null,
  model_used       varchar(255),
  created_at       timestamptz not null default now()
);

COMMENT ON TABLE lawschatter.messages
  IS '訊息表：儲存對話中的所有訊息（user/assistant/system/tool），並作為對話回放與列表統計的依據。';
COMMENT ON COLUMN lawschatter.messages.id
  IS '訊息唯一識別碼（UUID）。';
COMMENT ON COLUMN lawschatter.messages.conversation_id
  IS '所屬對話 ID（FK → conversations.id；刪除對話時一併刪除此對話的訊息）。';
COMMENT ON COLUMN lawschatter.messages.sender_role_code
  IS '訊息角色代碼（FK → roles.role_code），通常為 user/assistant/system/tool。';
COMMENT ON COLUMN lawschatter.messages.content
  IS '訊息內容（純文字）。';
COMMENT ON COLUMN lawschatter.messages.model_used
  IS '（選填）產生此訊息的模型代號（通常僅 assistant 會需要）。';
COMMENT ON COLUMN lawschatter.messages.created_at
  IS '建立時間（預設 now()），用於訊息串時間序與分頁。';

CREATE INDEX IF NOT EXISTS idx_messages_conv_time_seek
  ON lawschatter.messages (conversation_id, created_at, id);

COMMENT ON INDEX lawschatter.idx_messages_conv_time_seek
  IS '主查詢路徑：WHERE conversation_id = ? ORDER BY created_at, id；支援以 (created_at,id) 做游標分頁，避免 OFFSET 跳頁/重覆/漏資料。';


-- 需要 pg_trgm
create extension if not exists pg_trgm;

-- 模糊搜尋訊息內容
create index if not exists idx_messages_content_trgm
  on lawschatter.messages using gin (content gin_trgm_ops);
comment on index lawschatter.idx_messages_content_trgm
  is '使用 GIN + pg_trgm 加速訊息內容的模糊搜尋（LIKE/ILIKE/similarity）';


-- ========================================
-- Message Feedback（like/dislike + comment）
-- ========================================
create table if not exists lawschatter.message_feedback (
  id          uuid primary key default gen_random_uuid(),
  message_id  uuid not null references lawschatter.messages(id) on delete cascade,
  user_id     uuid not null references lawschatter.profiles(id) on delete cascade,
  rating      varchar(10) not null,  -- 'like' | 'dislike'
  comment     text,
  created_at  timestamptz not null default now(),
  constraint uq_feedback_unique_per_user_message unique (message_id, user_id),
  constraint ck_feedback_rating check (rating in ('like','dislike'))
);

-- 保留這個（依 user 查）
create index if not exists idx_message_feedback_user_id
  on lawschatter.message_feedback(user_id);

-- 可選：依訊息 + 時間排序
create index if not exists idx_feedback_message_created
  on lawschatter.message_feedback(message_id, created_at);

-- ========================================
-- User Settings（KV per user）
-- ========================================
create table if not exists lawschatter.user_settings (
  id          uuid primary key default gen_random_uuid(),
  user_id     uuid not null
              references lawschatter.profiles(id) on delete cascade,
  key         varchar(100) not null,
  value       jsonb not null,          -- 用 jsonb 更彈性
  updated_at  timestamptz not null default now(),
  constraint uq_user_settings unique (user_id, key)
);

drop trigger if exists trg_user_settings_updated_at on lawschatter.user_settings;
create trigger trg_user_settings_updated_at
before update on lawschatter.user_settings
for each row execute function lawschatter.touch_updated_at();

-- 若常用 jsonb 條件（例如 value @> '{"theme":"dark"}'），加速查詢
create index if not exists idx_user_settings_value_gin
  on lawschatter.user_settings using gin (value);

-- ========================================
-- Search Logs（可支援匿名：user_id 可為 NULL）
-- ========================================
create table if not exists lawschatter.search_logs (
  id              uuid primary key default gen_random_uuid(),
  user_id         uuid references lawschatter.profiles(id) on delete set null,
  query_text      text not null,
  created_at      timestamptz not null default now(),
  filters_applied text,
  result_count    int,
  constraint ck_search_logs_result_count_nonneg check (result_count is null or result_count >= 0),
  constraint ck_search_logs_query_nonempty check (length(trim(query_text)) > 0)
);

-- 表註釋
comment on table lawschatter.search_logs is '搜尋日誌表，用於記錄使用者或匿名訪客的查詢行為與結果資訊';

-- 欄位註釋
comment on column lawschatter.search_logs.id              is '搜尋紀錄唯一 ID（UUID）';
comment on column lawschatter.search_logs.user_id         is '執行搜尋的使用者 ID（可為 NULL 以支援匿名搜尋）';
comment on column lawschatter.search_logs.query_text      is '使用者輸入的搜尋關鍵字或自然語句';
comment on column lawschatter.search_logs.created_at      is '搜尋執行時間（UTC）';
comment on column lawschatter.search_logs.filters_applied is '套用的過濾條件摘要（可選）';
comment on column lawschatter.search_logs.result_count    is '當次搜尋返回的結果數量（可為 NULL）';

-- 索引
create index if not exists idx_search_logs_user_created
  on lawschatter.search_logs(user_id, created_at desc);

create index if not exists idx_search_logs_created_at
  on lawschatter.search_logs(created_at);

create extension if not exists pg_trgm;
create index if not exists idx_search_logs_query_trgm
  on lawschatter.search_logs using gin (query_text gin_trgm_ops);

-- 為索引加註解（注意這裡要加 lawschatter. 前綴）
comment on index lawschatter.idx_search_logs_user_created
  is '加速查詢特定使用者的搜尋紀錄，並依時間倒序排序（支援後台審核/統計）';

comment on index lawschatter.idx_search_logs_created_at
  is '加速依時間範圍篩選搜尋紀錄（支援維運清檔與報表）';

comment on index lawschatter.idx_search_logs_query_trgm
  is '使用 GIN + pg_trgm 加速搜尋 query_text 欄位的模糊搜尋（LIKE/ILIKE/similarity）';
